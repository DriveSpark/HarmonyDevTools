cmake_minimum_required(VERSION 3.16)
project(devtools_log_test LANGUAGES CXX)

set(capture "${CMAKE_CURRENT_LIST_DIR}/../../devtools/src/main/cpp")
add_executable(devtools_log_test log_capture_test.cpp "${capture}/log_capture.cpp" "${capture}/log_journal.cpp")
target_include_directories(devtools_log_test PRIVATE "${capture}")
target_compile_features(devtools_log_test PRIVATE cxx_std_17)
option(DEVTOOLS_TEST_LINUX_HOST "Use a HiLog callback stand-in for host Linux IPC tests" OFF)
if(DEVTOOLS_TEST_LINUX_HOST)
  target_include_directories(devtools_log_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/host")
  target_sources(devtools_log_test PRIVATE host_hilog.cpp)
  target_compile_definitions(devtools_log_test PRIVATE DEVTOOLS_TEST_LINUX_HOST)
  find_package(Threads REQUIRED)
  target_link_libraries(devtools_log_test PRIVATE Threads::Threads)
else()
  target_link_libraries(devtools_log_test PRIVATE libhilog_ndk.z.so)
endif()
